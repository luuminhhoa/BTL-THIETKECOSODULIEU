\section{Xác định Các Thực thể và Thuộc tính}
Dựa trên các phân hệ chức năng và quy tắc nghiệp vụ đã khảo sát, mô hình dữ liệu quan niệm của hệ thống OctaPoint CaaS xác định 11 thực thể cốt lõi cùng danh sách thuộc tính chi tiết.

Để đảm bảo tính liên kết chặt chẽ, kiến trúc dữ liệu của hệ thống được quy hoạch dựa trên hai trục trung tâm:
\begin{itemize}
    \item \textbf{Trục Doanh nghiệp (Merchant):} Đóng vai trò là gốc của cấu trúc đa khách thuê (Multi-tenant). Mọi dữ liệu về chiến dịch (Campaign), người dùng (User), cấu hình API (APICredential) đều được định tuyến và cô lập dựa trên khóa ngoại MerchantID.
    \item \textbf{Trục Sổ cái Giao dịch (Ledger):} Tập trung vào thực thể \texttt{CREDIT\_TRANSACTION}. Đây là rốn vũ trụ ghi nhận mọi biến động điểm thưởng, liên kết chặt chẽ với người dùng cuối (Customer), chiến dịch (Campaign) và chuỗi khối Sui để phục vụ đối soát.
\end{itemize}

\subsection{Phân hệ Tổ chức và Phân quyền (Tenant \& RBAC)}
\begin{itemize}
    \item \textbf{Thực thể TENANT (Đơn vị thuê bao gốc):}
    \begin{itemize}
        \item \underline{TenantID} (PK): Mã định danh duy nhất của công ty mẹ/chuỗi.
        \item TenantName (NOT NULL): Tên định danh gốc.
        \item SystemConfig (NULL): Cấu hình chung dạng JSON.
    \end{itemize}

    \item \textbf{Thực thể MERCHANT (Doanh nghiệp đối tác):}
    \begin{itemize}
        \item \underline{MerchantID} (PK): Mã định danh cửa hàng/chi nhánh.
        \item \textit{TenantID} (FK, NOT NULL): Liên kết tới bảng Tenant.
        \item BusinessName (NOT NULL): Tên thương hiệu/cửa hàng.
        \item Status (NOT NULL): Trạng thái hoạt động (Active/Suspended).
    \end{itemize}

    \item \textbf{Thực thể USER (Người dùng/Nhân viên):}
    \begin{itemize}
        \item \underline{UserID} (PK): Mã định danh nhân viên.
        \item \textit{MerchantID} (FK, NOT NULL): Mã cửa hàng trực thuộc.
        \item \textit{RoleID} (FK, NOT NULL): Mã phân quyền.
        \item \textit{ManagerID} (FK, NULL): Mã người quản lý trực tiếp (tự tham chiếu).
        \item Email (UNIQUE): Email đăng nhập hệ thống.
        \item PasswordHash (NOT NULL): Mật khẩu đã được mã hóa.
    \end{itemize}

    \item \textbf{Thực thể ROLE (Vai trò):}
    \begin{itemize}
        \item \underline{RoleID} (PK): Mã định danh vai trò.
        \item RoleName (NOT NULL): Tên vai trò (Admin, Manager, Staff).
    \end{itemize}
\end{itemize}

\subsection{Phân hệ Tín dụng và Điểm thưởng (Loyalty Engine)}
\begin{itemize}
    \item \textbf{Thực thể CAMPAIGN (Chiến dịch điểm thưởng):}
    \begin{itemize}
        \item \underline{CampaignID} (PK): Mã định danh chiến dịch.
        \item \textit{TenantID} (FK, NOT NULL): Chuỗi (Tenant) sở hữu chiến dịch.
        \item CampaignName (NOT NULL): Tên chiến dịch.
        \item StartDate (NOT NULL): Ngày bắt đầu.
        \item EndDate (NULL): Thời điểm kết thúc.
        \item Status (NOT NULL): Trạng thái (Draft/Active/Expired).
    \end{itemize}

    \item \textbf{Thực thể CAMPAIGN\_MERCHANT (Phạm vi áp dụng):}
    \begin{itemize}
        \item \underline{\textit{CampaignID}} (PK, FK): Tham chiếu chiến dịch.
        \item \underline{\textit{MerchantID}} (PK, FK): Tham chiếu cửa hàng được áp dụng.
    \end{itemize}

    \item \textbf{Thực thể REWARD\_RULE (Quy tắc tính điểm):}
    \begin{itemize}
        \item \underline{RuleID} (PK): Mã định danh quy tắc.
        \item \textit{CampaignID} (FK, NOT NULL): Thuộc chiến dịch nào.
        \item ActionType (NOT NULL): Loại hành động.
        \item PointsMultiplier (NOT NULL): Hệ số nhân điểm.
    \end{itemize}

    \item \textbf{Thực thể CUSTOMER (Khách hàng):}
    \begin{itemize}
        \item \underline{CustomerID} (PK): Mã định danh khách hàng nội bộ.
        \item WalletAddress (UNIQUE, NULL): Địa chỉ blockchain (Sui).
        \item Email (NULL): Địa chỉ email.
    \end{itemize}

    \item \textbf{Thực thể CREDIT (Số dư ví điểm chung):}
    \begin{itemize}
        \item \underline{CreditID} (PK): Mã định danh bản ghi số dư.
        \item \textit{CustomerID} (FK, NOT NULL): Liên kết khách hàng.
        \item \textit{TenantID} (FK, NOT NULL): Thuộc chuỗi (Tenant) nào.
        \item AvailableBalance (NOT NULL): Số dư khả dụng hiện tại.
        \item \textit{* Ràng buộc hệ thống:} \texttt{UNIQUE(CustomerID, TenantID)} để đảm bảo một khách hàng chỉ có một ví tại một chuỗi.
    \end{itemize}
\end{itemize}

\subsection{Phân hệ Sổ cái Giao dịch và Kiểm toán (Ledger \& Audit)}
\begin{itemize}
    \item \textbf{Thực thể CREDIT\_TRANSACTION (Giao dịch điểm):}
    \begin{itemize}
        \item \underline{TransactionID} (PK): Mã định danh giao dịch nội bộ.
        \item \textit{CreditID} (FK, NOT NULL): Liên kết tới ví số dư.
        \item \textit{CampaignID} (FK, NULL): Phát sinh từ chiến dịch nào.
        \item \textit{MerchantID} (FK, NOT NULL): Cửa hàng nơi phát sinh giao dịch.
        \item Amount (NOT NULL): Số điểm biến động.
        \item TransactionType (NOT NULL): Loại (Earn/Redeem/Refund).
        \item SuiTxDigest (UNIQUE, NULL): Mã băm đối soát trên chuỗi khối Sui.
        \item CreatedAt (NOT NULL): Thời điểm giao dịch.
    \end{itemize}
    
    \item \textbf{Thực thể API\_CREDENTIAL (Khóa tích hợp):}
    \begin{itemize}
        \item \underline{KeyID} (PK): Mã định danh khóa.
        \item \textit{MerchantID} (FK, NOT NULL): Doanh nghiệp/Cửa hàng sở hữu khóa.
        \item ApiKey (NOT NULL, UNIQUE): Mã định danh khóa công khai.
        \item SecretHash (NOT NULL): Mã băm của Secret Key.
    \end{itemize}
\end{itemize}
